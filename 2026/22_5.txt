from pathlib import Path


archivo_fasta = Path(__file__).with_name("gene.fna")
nombre_organismo = archivo_fasta.stem

# Leer y separar las secuencias del archivo FASTA.
with open(archivo_fasta, "r", encoding="utf-8") as archivo:
    lineas = archivo.readlines()

secuencias = []
encabezado = None
bases_secuencia = []

for linea in lineas:
    contenido = linea.strip()

    if not contenido:
        continue

    if contenido.startswith(">"):
        if encabezado is not None:
            secuencias.append((encabezado, "".join(bases_secuencia)))

        encabezado = contenido[1:].strip()
        bases_secuencia = []
    else:
        if encabezado is None:
            raise ValueError("El archivo tiene bases antes de su primer encabezado.")
        bases_secuencia.append(contenido.upper())

if encabezado is not None:
    secuencias.append((encabezado, "".join(bases_secuencia)))

if not secuencias:
    raise ValueError("El archivo FASTA no contiene secuencias.")

# 1. Imprimir el nombre del archivo y cada encabezado, sin el signo >.
for encabezado, secuencia in secuencias:
    print(f"Organismo {nombre_organismo}, contig <{encabezado}>")

# 2. Imprimir la fracción GC de cada línea de secuencia.
# El número corresponde a la línea real del archivo, contando también encabezados.
for numero_linea, linea in enumerate(lineas, start=1):
    contenido = linea.strip().upper()

    if not contenido or contenido.startswith(">"):
        continue

    bases_validas = [base for base in contenido if base in "ACGT"]
    if bases_validas:
        cantidad_gc = sum(base in "GC" for base in bases_validas)
        fraccion_gc = cantidad_gc / len(bases_validas)
        print(f"GC línea {numero_linea}: {fraccion_gc:.2f}")
    else:
        print(f"GC línea {numero_linea}: no hay bases A, C, G o T")

# 3. Imprimir la secuencia complementaria completa de cada contig.
# Las líneas de secuencia de un mismo contig se unen antes de procesarlas.
complementos = {"A": "T", "T": "A", "C": "G", "G": "C"}

for encabezado, secuencia in secuencias:
    secuencia_complementaria = ""
    for base in secuencia:
        secuencia_complementaria += complementos.get(base, base)
    print(f"Complementaria de <{encabezado}>: {secuencia_complementaria}")

# 3 adicional. Transcribir cada secuencia de ADN a ARN (T se reemplaza por U).
for encabezado, secuencia in secuencias:
    secuencia_arn = secuencia.replace("T", "U")
    print(f"ARN de <{encabezado}>: {secuencia_arn}")

# 4. Imprimir las líneas que contienen el codón ATG.
# Se convierte a mayúsculas para encontrar también un "atg" escrito en minúsculas.
for numero_linea, linea in enumerate(lineas, start=1):
    contenido = linea.strip().upper()

    if contenido and not contenido.startswith(">") and "ATG" in contenido:
        print(f"ATG está en línea {numero_linea}")

# 5. Calcular el contenido GC total y la composición de cada base.
conteos = {"A": 0, "C": 0, "G": 0, "T": 0}

for encabezado, secuencia in secuencias:
    for base in secuencia:
        if base in conteos:
            conteos[base] += 1

total_bases = sum(conteos.values())
if total_bases == 0:
    raise ValueError("El archivo no contiene bases A, C, G o T para calcular porcentajes.")

fraccion_gc_total = (conteos["G"] + conteos["C"]) / total_bases
print(f"GC total del archivo: {fraccion_gc_total:.2f}")

for base in "ATCG":
    porcentaje = conteos[base] / total_bases * 100
    print(f"Porcentaje de {base}: {porcentaje:.2f}%")

porcentaje_at = (conteos["A"] + conteos["T"]) / total_bases * 100
print(f"Porcentaje de A + T: {porcentaje_at:.2f}%")
